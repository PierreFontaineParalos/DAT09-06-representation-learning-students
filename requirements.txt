# IMPORTANT — CPU-only PyTorch:
#   On Windows and macOS, the default PyPI wheels are already CPU-only.
#   On Linux, the default wheels drag in ~2.5 GB of CUDA libraries you do not
#   need. Linux users should install with the CPU index instead:
#
#       pip install -r requirements.txt --extra-index-url https://download.pytorch.org/whl/cpu
#
# Install:
#       python -m venv .venv
#       .venv\Scripts\activate          (Windows)   |   source .venv/bin/activate
#       pip install --upgrade pip
#       pip install -r requirements.txt
